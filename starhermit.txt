# StarHermit engine script game: one package, played natively by the
# StarHermit client and in browsers by the engine's web runtime. Browsers open
# launch=index.html, the loader `npm run package` writes (it pins the engine
# on wiki.starhermit.com by hash; never edit it). Re-run `npm run package`
# after every change and commit index.html + starhermit-files.json.
launch=index.html
engine.api=1
engine.entry=native/main.ts
engine.version=1.23.0
engine.minor=23
engine.capabilities=platform,storage,clipboard
# Ultra's 512-px card faces (28 on the board, more as the waste turns) need more than the default 56 MiB.
engine.limits.image_mib=160

# Keyboard actions (players rebind keys and pads in the runtime's Options).
# The tableau is also played with the pointer or a finger: a press on an
# uncovered card selects it, a second card totalling 14 removes the pair, a
# press on the stock draws.
engine.group.prev=Cards
control.prev=ArrowLeft+ArrowUp | Previous card
control.next=ArrowRight+ArrowDown | Next card
control.confirm=Enter | Select
control.cancel=Escape | Cancel / pause
engine.group.draw=Round
control.draw=KeyD | Draw
control.recycle=KeyR | Recycle
control.undo=KeyU | Undo
control.hint=KeyH | Hint
control.camera=KeyC | Reset camera

# Gamepad defaults (the browser build had none): the D-pad or left stick
# moves between the cards you can play, South selects, East cancels or
# pauses (Start pauses), North draws, West undoes, the shoulders recycle
# and hint, the right stick press resets the camera.
engine.gamepad.prev=DpadLeft+DpadUp+LeftStickLeft+LeftStickUp
engine.gamepad.next=DpadRight+DpadDown+LeftStickRight+LeftStickDown
engine.gamepad.confirm=South
engine.gamepad.cancel=East+Start
engine.gamepad.draw=North
engine.gamepad.undo=West
engine.gamepad.recycle=RightShoulder
engine.gamepad.hint=LeftShoulder
engine.gamepad.camera=RightStickPress
